#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>

// Definiciones de la pantalla OLED
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// Definición del Sensor DHT11
#define DHTPIN 2       // Pin GPIO donde está conectado el DATA del DHT11
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

// Definición del Módulo de Relé
#define RELAY_PIN 0    // Pin GPIO conectado al IN/SIG del relé

// Umbral de temperatura para encender el ventilador (en grados Celsius)
const float TEMP_UMBRAL = 28.0; 

void setup() {
  Serial.begin(115200);
  
  // Inicializar el bus I2C indicando los pines SDA y SCL (GPIO 4 y GPIO 5)
  Wire.begin(4, 5);
  
  // Inicializar sensor DHT
  dht.begin();
  
  // Configurar pin del relé como salida
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // Arrancar con el ventilador apagado

  // Inicializar pantalla OLED (puedes cambiar a 0x3D si con 0x3C no enciende)
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("No se encontró la pantalla OLED"));
    for(;;);
  }
  
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println(F("O-Hare Air Iniciando..."));
  display.display();
  delay(2000);
}

void loop() {
  // Esperar unos segundos entre mediciones
  delay(2000);

  // Leer humedad y temperatura
  float h = dht.readHumidity();
  float t = dht.readTemperature(); // En grados Celsius

  // Comprobar si hubo un error en la lectura
  if (isnan(h) || isnan(t)) {
    Serial.println(F("Error al leer el sensor DHT11!"));
    display.clearDisplay();
    display.setCursor(0, 0);
    display.println(F("Error sensor DHT11"));
    display.display();
    return;
  }

  // Control del ventilador según la temperatura
  if (t >= TEMP_UMBRAL) {
    digitalWrite(RELAY_PIN, HIGH); // Enciende el relé (ventilador ON)
  } else {
    digitalWrite(RELAY_PIN, LOW);  // Apaga el relé (ventilador OFF)
  }

  // Mostrar datos en la pantalla OLED
  display.clearDisplay();
  
  // Título
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println(F("--- O-HARE AIR ---"));

  // Temperatura
  display.setCursor(0, 20);
  display.print(F("Temp: "));
  display.print(t);
  display.println(F(" C"));

  // Humedad
  display.setCursor(0, 35);
  display.print(F("Humedad: "));
  display.print(h);
  display.println(F(" %"));

  // Estado del Ventilador
  display.setCursor(0, 50);
  display.print(F("Ventilador: "));
  if (t >= TEMP_UMBRAL) {
    display.println(F("ENCENDIDO"));
  } else {
    display.println(F("APAGADO"));
  }

  display.display();

  // Depuración por puerto serie
  Serial.print(F("Temp: ")); Serial.print(t); Serial.print(F(" C  | "));
  Serial.print(F("Humedad: ")); Serial.print(h); Serial.print(F(" %  | "));
  Serial.print(F("Ventilador: ")); Serial.println(t >= TEMP_UMBRAL ? "ON" : "OFF");
}
